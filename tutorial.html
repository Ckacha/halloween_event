Building Ghost Hunt

A ghost-clicking arcade game in about 50 lines of p5.js. Follow the steps in order — each one adds a working piece, so you can stop and have something playable after any of them.

Step 0

Set up your sketch

Open editor.p5js.org, clear the default code, and paste this. p5.js calls setup() once and draw() on every frame — that's the whole engine.

sketch.js
function setup() {
  createCanvas(640, 480);
}

function draw() {
  background(20);
}
Hit the ▶ Play button. You should see a dark grey canvas — that's your game screen.
Step 1

Draw a ghost

Start with a shape you can reason about — a pale circle. You can make it look more ghost-like later; the click logic doesn't care what it looks like.

sketch.js
function draw() {
  background(20);
  drawGhost(width / 2, height / 2);
}

function drawGhost(x, y) {
  noStroke();
  fill(200, 255, 220);
  ellipse(x, y, 60, 60);
}
Step 2

Spawn it, again and again

A ghost object holding its own position, and a spawn function that picks a new random one. Checking frameCount against an interval is the cheapest way to do "every N frames" without a separate timer.

sketch.js
let ghost;
let spawnEvery = 60; // ~1 second at 60fps

function setup() {
  createCanvas(640, 480);
  spawnGhost();
}

function draw() {
  background(20);
  if (frameCount % spawnEvery === 0) spawnGhost();
  drawGhost(ghost.x, ghost.y);
}

function spawnGhost() {
  ghost = { x: random(40, width - 40), y: random(40, height - 40) };
}

function drawGhost(x, y) {
  noStroke();
  fill(200, 255, 220);
  ellipse(x, y, 60, 60);
}
Step 3

Click to score

p5.js calls mousePressed() on every click. dist() gives you the distance between the click and the ghost — if it's less than the ghost's radius, it's a hit.

sketch.js
let score = 0;

function mousePressed() {
  let hit = dist(mouseX, mouseY, ghost.x, ghost.y) < 30;
  if (hit) {
    score++;
    spawnGhost();
  }
}

function draw() {
  background(20);
  if (frameCount % spawnEvery === 0) spawnGhost();
  drawGhost(ghost.x, ghost.y);

  fill(255);
  textSize(20);
  text('Score: ' + score, 16, 30);
}
Try it: click the ghost a few times. Score should go up and the ghost should jump to a new spot.
Step 4

Add a timer

millis() returns milliseconds since the sketch started — subtract a saved start time from it to get elapsed seconds, and count down from there.

sketch.js
let duration = 30;
let startTime;

function setup() {
  createCanvas(640, 480);
  startTime = millis();
  spawnGhost();
}

function draw() {
  background(20);

  let elapsed = (millis() - startTime) / 1000;
  let timeLeft = max(0, duration - elapsed);

  if (frameCount % spawnEvery === 0) spawnGhost();
  drawGhost(ghost.x, ghost.y);

  fill(255);
  textSize(20);
  text('Score: ' + score, 16, 30);
  text('Time: ' + ceil(timeLeft), width - 110, 30);
}
Step 5

Start & game-over screens

One variable, state, decides what the screen shows and what a click does. This is the same pattern the reference build uses — it just has a third state's worth of decoration.

sketch.js
let state = 'start'; // 'start' | 'play' | 'over'

function draw() {
  if (state === 'start') {
    background(20);
    textAlign(CENTER, CENTER);
    textSize(28);
    fill(255);
    text('Ghost Hunt\nClick to start', width / 2, height / 2);
    return;
  }

  if (state === 'over') {
    background(20);
    textAlign(CENTER, CENTER);
    textSize(24);
    fill(255);
    text('Game over\nScore: ' + score, width / 2, height / 2);
    return;
  }

  // state === 'play'
  background(20);
  let elapsed = (millis() - startTime) / 1000;
  let timeLeft = max(0, duration - elapsed);
  if (timeLeft <= 0) { state = 'over'; return; }

  if (frameCount % spawnEvery === 0) spawnGhost();
  drawGhost(ghost.x, ghost.y);

  textAlign(LEFT, BASELINE);
  fill(255);
  textSize(20);
  text('Score: ' + score, 16, 30);
  text('Time: ' + ceil(timeLeft), width - 110, 30);
}

function mousePressed() {
  if (state === 'start') {
    state = 'play';
    startTime = millis();
    score = 0;
    spawnGhost();
    return;
  }
  if (state === 'over') {
    state = 'start';
    return;
  }
  if (dist(mouseX, mouseY, ghost.x, ghost.y) < 30) {
    score++;
    spawnGhost();
  }
}
That's a complete Ghost Hunt. Start screen, click-to-score, countdown, game over, restart. Everything past here is optional.
Next

Level up: 5 upgrades

Short, self-contained additions to the base game above. Pick the ones that sound fun — the reference build uses all five plus a couple more.

  • Give the ghost movement

    ~10 min

    Store a velocity on the ghost and nudge its position every frame, bouncing off the walls.

    ghost.vx = random(-2, 2);
ghost.vy = random(-2, 2);

// in draw(), before drawGhost():
ghost.x += ghost.vx;
ghost.y += ghost.vy;
if (ghost.x < 30 || ghost.x > width - 30) ghost.vx *= -1;
if (ghost.y < 30 || ghost.y > height - 30) ghost.vy *= -1;

  • Add a second enemy type

    ~20 min

    Swap the single ghost variable for an array of entities, each with a type. Spawn and draw branch on that type.

    let entities = [];

function spawnGhost() {
  let type = random() < 0.7 ? 'ghost' : 'pumpkin';
  entities.push({
    type: type,
    x: random(40, width - 40),
    y: random(40, height - 40)
  });
}

// in mousePressed(), loop entities instead of checking one ghost:
for (let i = entities.length - 1; i >= 0; i--) {
  let e = entities[i];
  if (dist(mouseX, mouseY, e.x, e.y) < 30) {
    score += e.type === 'pumpkin' ? 25 : 10;
    entities.splice(i, 1);
    break;
  }
}

  • Sound effects

    ~10 min

    The easy path is p5.sound: load a short clip and play it on a hit. Add the p5.sound library in the editor's sketch settings first.

    let hitSound;

function preload() {
  hitSound = loadSound('hit.mp3');
}

// on a successful click:
hitSound.play();
    No audio file handy? The reference build synthesizes its beeps with the Web Audio API instead — no files needed, just an oscillator and a quick volume fade.

  • Particle burst on a hit

    ~15 min

    An array of tiny dots with random outward velocity, each fading out and removing itself once its life hits zero.

    let particles = [];

function burst(x, y) {
  for (let i = 0; i < 10; i++) {
    let a = random(TWO_PI);
    particles.push({ x: x, y: y, vx: cos(a) * 2, vy: sin(a) * 2, life: 1 });
  }
}

// in draw(), after drawing entities:
for (let i = particles.length - 1; i >= 0; i--) {
  let pt = particles[i];
  pt.x += pt.vx; pt.y += pt.vy; pt.life -= 0.04;
  if (pt.life <= 0) { particles.splice(i, 1); continue; }
  noStroke();
  fill(200, 255, 220, pt.life * 255);
  ellipse(pt.x, pt.y, 5);
}

  • Difficulty ramp

    ~10 min

    Shrink the spawn interval as the run goes on, using lerp() between a slow start and a fast finish.

    // in draw(), replacing the fixed spawnEvery:
let progress = constrain(elapsed / duration, 0, 1);
let interval = lerp(60, 20, progress); // frames between spawns
if (frameCount % floor(interval) === 0) spawnGhost();
